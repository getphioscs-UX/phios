export const escWorld=value=>String(value??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
export async function queryWorldExplanation(question,locale,fetcher=fetch){try{const r=await fetcher('/api/world-question-explanation',{method:'POST',headers:{'content-type':'application/json'},body:JSON.stringify({question,locale}),cache:'no-store'});if(!r.ok)return null;const data=await r.json();return data.ok===true?data.explanation:null;}catch{return null;}}
export function worldCustomerAnswerHTML(answer,locale='en'){
 const zh=locale==='zh-Hans',t=(z,e)=>zh?z:e,esc=escWorld,x=answer.explanation;
 const direct=x?.sources?.length?`${t('现有资料支持以下部分解释：','The available material supports this part of the explanation: ')}${x.directAnswer}`:(x?.directAnswer||answer.directAnswer);
 const reasons=x?.reasons?.length?x.reasons:(answer.mechanism||[]);
 const next=x?.nextSteps||[t('明确你想比较的地点、时期和条件。','Identify the place, period and conditions you want to compare.'),t('把与你有关的条件带回自己的问题。','Bring the conditions that matter into your own question.')];
 const sources=x?.sources||[],observations=answer.observations||[];
 return `<h2>${t('先回答你的问题','Your question, answered first')}</h2><p data-customer-direct>${esc(direct)}</p>${reasons.length?`<h3>${t('为什么会这样','Why this can happen')}</h3>${reasons.map(s=>`<p>${esc(s)}</p>`).join('')}`:''}<h3>${t('与现实生活的关系','What this means in everyday life')}</h3><p>${esc(x?.impact||t('这些总体资料可以帮助理解背景；个人影响仍要对照自己的收入、支出、时间和选择条件。','Aggregate evidence helps explain context; personal effects still depend on your income, costs, time and options.'))}</p><h3>${t('可以继续了解什么','Where to go next')}</h3><ul>${next.map(s=>`<li>${esc(s)}</li>`).join('')}</ul><p class="small">${esc(x?.boundary||t('这是注明时期的资料，不代表实时情况，也不能直接推出个人结果。','This evidence refers to a stated period; it is not real-time and does not establish an individual outcome.'))}</p><details data-source-details><summary>${t('查看来源、定义与适用范围','Sources, definitions and scope')}</summary>${sources.map(s=>`<p><a href="${esc(s.href)}">${esc(s.title)}</a></p><blockquote>${esc(s.text)}</blockquote>`).join('')}${observations.map(o=>`<p><a href="${esc(o.sourceUrl)}">${esc(o.sourcePublisher)} · ${esc(o.observationPeriod)}</a></p><p>${esc(o.measurementDefinition)}</p>`).join('')}<ul>${[...(answer.missingEvidence||[]),...(answer.limits||[])].map(l=>`<li>${esc(l)}</li>`).join('')}</ul><details data-developer-details><summary>${t('开发诊断','Developer diagnostics')}</summary><pre style="white-space:pre-wrap;overflow-wrap:anywhere">${esc(JSON.stringify({sources,observations,state:answer.answerState},null,2))}</pre></details></details>`;
}

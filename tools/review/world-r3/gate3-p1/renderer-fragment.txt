function renderEvidenceAnswer(){
 if(!evidenceIntent||state.space==='me'){answerSequence++;answerSignature='';answerHost.hidden=true;answerHost.replaceChildren();return;}
 const plan=worldQuestionPlan(customQuestion,{intent:evidenceIntent,region:state.region,year:state.year,period:String(state.year)+(evidenceQuarter?'-'+evidenceQuarter:'')});
 const signature=JSON.stringify([plan.intent,plan.scope,state.lang,customQuestion]);if(signature===answerSignature)return;answerSignature=signature;const seq=++answerSequence;answerHost.hidden=false;answerHost.innerHTML=`<p role="status">${pair('正在核对指定范围的来源…','Checking evidence for the selected scope…')}</p>`;
 Promise.all(plan.requests.map(request=>queryWorldEvidence(request))).then(evidence=>{
  if(seq!==answerSequence)return;const answer=composeWorldAnswer(plan,evidence,getLocale());answerControllerResult=answer;answerHost.dataset.answerState=answer.answerState;answerHost.dataset.intent=answer.intent;
  const value=n=>new Intl.NumberFormat(getLocale(),{maximumFractionDigits:2}).format(n);
  const unit=u=>({'reported_records':pair('申报行动记录','reported action records'),'respondent_banks':pair('受访银行家数','respondent banks'),'index_2010_100':pair('指数（2010=100）','index (2010=100)'),'MYR_per_month':pair('令吉／月（调查参考年）','MYR/month (survey reference year)')})[u]||pair('数值','value');
  const metric=o=>o.actionCode?({1:pair('已发放','Originated'),2:pair('批准后未接受','Approved, not accepted'),3:pair('拒绝','Denied')})[o.actionCode]:o.seriesId==='MY_CPI'?pair('总体消费价格指数','Overall consumer price index'):o.seriesId==='MY_INCOME'?pair('家庭月收入中位数','Median monthly household income'):o.surveyMeasure==='standards'?pair('收紧标准的银行','Banks tightening standards'):pair('需求增强的银行','Banks reporting stronger demand');
  const lines=items=>items.map(x=>`<p>${esc(x)}</p>`).join('');
  const links=[...new Map(answer.relatedSources.filter(s=>safeWorldSource(s.href)).map(s=>[s.href,s])).values()];
  answerHost.innerHTML=`<h2>${pair('针对问题的回答','An answer to your question')}</h2>${customQuestion?`<p>${esc(customQuestion)}</p>`:''}<p class="small muted">${esc(nameGeo(state.region))} · ${esc(answer.period)} · ${pair('有界规则与来源计算，无模型生成','Bounded rules and source calculations; no model generation')}</p><p data-world-direct-answer style="font-size:1.15rem;font-weight:600">${esc(answer.directAnswer)}</p>${answer.mechanism.length?`<h3>${pair('如何理解','How to interpret this')}</h3>${lines(answer.mechanism)}`:''}${answer.observations.length?`<details><summary>${pair('查看实值、定义与来源','Inspect values, definitions and sources')} (${answer.observations.length})</summary>${answer.observations.map(o=>`<div style="border-top:1px solid #b7c8c5;padding:12px 0"><strong>${esc(metric(o))}: ${esc(value(o.value))} ${esc(unit(o.unit))}</strong><p>${pair('观察期','Observed')}: ${esc(o.observationPeriod)} · ${pair('发布日','Published')}: ${esc(o.publishedAt||pair('未保留','not retained'))}</p><p>${pair('取得时间','Retrieved')}: ${esc(o.retrievedAt)}</p><a href="${esc(safeWorldSource(o.sourceUrl)||'#')}" target="_blank" rel="noopener noreferrer">${esc(o.sourcePublisher)} · ${pair('原始来源','Original source')}</a><details><summary>${pair('追溯版本与适用范围','Trace version and scope')}</summary><p style="overflow-wrap:anywhere">SHA-256: ${esc(o.rawArtifactHash)}</p><p>${esc(o.sourceDataset)} · ${esc(o.transformationId)}</p><p>${esc(o.licenseState==='CC_BY_4_0'?pair('来源：马来西亚统计局。知识共享署名4.0；已选择及整理数据。','Source: DOSM. CC BY4.0; data selected and formatted.'):pair('美国官方公共数据；非第三方资料，不代表官方背书。','US official public data; third-party content excluded; no endorsement.'))}</p></details></div>`).join('')}</details>`:''}${answer.derivedMetrics.length?`<h3>${pair('计算口径','Calculation scope')}</h3>${answer.derivedMetrics.map(d=>`<p>${esc(d.metric==='denial_share_in_defined_hmda_scope'?pair('购房用途行动记录拒绝占比','Home-purchase action-record denial share'):d.metric==='net_tightening'?pair('银行净收紧标准','Net tightening by banks'):pair('银行净需求增强','Net stronger demand by banks'))}: ${esc(value(d.numerator))} / ${esc(value(d.denominator))}${d.metric==='denial_share_in_defined_hmda_scope'?'':' × 100'}</p>`).join('')}`:''}${answer.missingEvidence.length?`<h3>${pair('还缺什么证据','What evidence is missing')}</h3>${lines(answer.missingEvidence)}`:''}${answer.limits.length?`<h3>${pair('适用边界','Limits')}</h3>${lines(answer.limits)}`:''}${links.length?`<h3>${pair('核对来源','Check the sources')}</h3><ul>${links.map(s=>`<li><a href="${esc(safeWorldSource(s.href))}" target="_blank" rel="noopener noreferrer">${esc(s.label)}${s.period?' · '+esc(s.period):''}</a></li>`).join('')}</ul>`:''}<p class="small muted">${pair('原问题只保留在本页内存；刷新后清除原文，公共地区、年份与意图仍保留。','Original wording stays in page memory and clears on refresh; public region, period and intent remain.')}</p><div class="actions">${button('pane',pair('继续手动提问','Continue with a manual question'),'data-pane="ask"')}</div>`;
  if(answer.answerState==='ANSWERED_BOUNDED')answerHost.setAttribute('role','region');else answerHost.setAttribute('role','status');
 }).catch(()=>{if(seq===answerSequence){answerHost.dataset.answerState='SERVICE_ERROR';answerHost.textContent=pair('证据服务暂不可用；未生成数值。','The evidence service is unavailable; no value was generated.');}});
}
